**Goal:** Replace payleor.com's pages with the approved Landing B site: index, Payleor Network, Discover, Get found, four audience pages, pricing, contact and a blog, with mega-menu navigation and a full search and AI visibility layer.

**Architecture:** Pages live in `apps/web/src/app/(site)/`. import {
```ts
  type BillingInterval, bundleQuote, getPlan, getSellerPlan, NETWORK_FEE_BPS, type PlanId, platformFeeMinor, type SellerPlanId,
} from "@jendal-workspace/core";

export type Sale =
  | { kind: "services"; planId: SellerPlanId; amountMinor: number; source: "own" | "network" }
  | { kind: "store"; planId: PlanId; amountMinor: number; source: "own" | "network" };

export type SaleFees = { feeBps: number; feeMinor: number; networkMinor: number; totalMinor: number };

// What Payleor takes on one sale, with the same rounding the ledger uses. The network fee applies to a services
// client's first project only (pricing page, 2026-10-19); store orders never carry it until that is decided.
export function saleFees(sale: Sale): SaleFees {
  const feeBps = sale.kind === "services" ? getSellerPlan(sale.planId).feeBps : getPlan(sale.planId).orderFeeBps;
  if (sale.amountMinor <= 0) return { feeBps, feeMinor: 0, networkMinor: 0, totalMinor: 0 };
  const feeMinor = platformFeeMinor(sale.amountMinor, feeBps);
  const networkMinor = sale.kind === "services" && sale.source === "network" ? platformFeeMinor(sale.amountMinor, NETWORK_FEE_BPS) : 0;
  return { feeBps, feeMinor, networkMinor, totalMinor: feeMinor + networkMinor };
}

export function bundleMonthly(services: SellerPlanId, store: PlanId, interval: BillingInterval) {
  const q = bundleQuote(getSellerPlan(services), getPlan(store), interval);
  return { totalCents: q.totalCents, halved: q.discountedProduct, feeBps: q.feeBps };
}
```

```ts
    description: p.description,
    alternates: { canonical: abs(p.path) },
    openGraph: { title: p.title, description: p.description, url: abs(p.path), siteName: "Payleor", type: p.kind ?? "website" },

export const organization = () =>

  ({

    potentialAction: { "@type": "SearchAction", target: `${SITE_ORIGIN}/discover?q={search_term_string}`, "query-input": "required name=search_term_string" },

export function breadcrumbs(trail: ReadonlyArray<readonly [name: string, path: string]>) {
```

```ts
import type { Listing } from "@/content/site/listings";

export type Filters = { q?: string; category?: Listing["category"] | "all"; type?: Listing["kind"] | "all"; done?: Listing["done"] | "all"; sort?: "featured" | "price" };

export function browse(items: readonly Listing[], f: Filters): Listing[] {
  const words = (f.q ?? "").toLowerCase().split(/\s+/).filter(Boolean);
  const hit = (i: Listing) => {
    const text = `${i.title} ${i.seller} ${i.kind} ${i.done} ${i.category} ${i.detail}`.toLowerCase();
    return words.every((w) => text.includes(w));
  };
  const out = items.filter((i) =>
    (!f.category || f.category === "all" || i.category === f.category) &&
    (!f.type || f.type === "all" || i.kind === f.type) &&
    (!f.done || f.done === "all" || i.done === f.done) && hit(i));
  return f.sort === "price" ? out.toSorted((a, b) => a.priceDollars - b.priceDollars) : out;
}
```
